local function queueScriptReload()
    local queueFunc = (syn and syn.queue_on_teleport)
        or queue_on_teleport
        or (Fluxus and Fluxus.queue_on_teleport)
        or (teleport and teleport.queue)

    if queueFunc then
        pcall(function()
            queueFunc([[
                repeat task.wait(1) until game:IsLoaded()
                task.wait(2)
                local executed = false
                if isfile then
                    if isfile("auto_farm.lua") then
                        loadstring(readfile("auto_farm.lua"))()
                        executed = true
                    elseif isfile("auto_farm") then
                        loadstring(readfile("auto_farm"))()
                        executed = true
                    end
                end
                if not executed and loadfile then
                    pcall(function() loadfile("auto_farm.lua")() executed = true end)
                    if not executed then
                        pcall(function() loadfile("auto_farm")() end)
                    end
                end
                print("Storm Hub: Local 'auto_farm' reloaded successfully after teleport/rejoin!")
            ]])
        end)
    end
end

local function toggleInvisible(state)
    isInvis = state
    if invisConnection then 
        invisConnection:Disconnect() 
        invisConnection = nil 
    end

    if state then
        invisConnection = RunService.Heartbeat:Connect(function()
            if rootPart and character and character:FindFirstChildOfClass("Humanoid") then
                local realCFrame = rootPart.CFrame
                local fakeCFrame = realCFrame * CFrame.new(0, -200000, 0)

                rootPart.CFrame = fakeCFrame
                character:FindFirstChildOfClass("Humanoid").CameraOffset = fakeCFrame:ToObjectSpace(CFrame.new(realCFrame.Position)).Position

                RunService.RenderStepped:Wait()
                if rootPart then 
                    rootPart.CFrame = realCFrame 
                end
            end
        end)
        sendNotification({ Title = "Invisibility Active", Desc = "Your character is hidden from other players." })
    else
        if character and character:FindFirstChildOfClass("Humanoid") then
            character:FindFirstChildOfClass("Humanoid").CameraOffset = Vector3.zero
        end
        sendNotification({ Title = "Invisibility Disabled", Desc = "Your character is visible again." })
    end
end

local function cleanupAlignFarm()
    if alignPos then pcall(function() alignPos:Destroy() end) alignPos = nil end
    if alignRot then pcall(function() alignRot:Destroy() end) alignRot = nil end
    if farmAttachment then pcall(function() farmAttachment:Destroy() end) farmAttachment = nil end
end

local function moveToCoinAlign(targetPos)
    if not rootPart then return end

    if not farmAttachment or not farmAttachment.Parent then
        cleanupAlignFarm()
        farmAttachment = Instance.new("Attachment", rootPart)

        alignPos = Instance.new("AlignPosition")
        alignPos.Mode = Enum.PositionAlignmentMode.OneAttachment
        alignPos.Attachment0 = farmAttachment
        alignPos.RigidityEnabled = true
        alignPos.Position = rootPart.Position
        alignPos.Parent = rootPart

        alignRot = Instance.new("AlignOrientation")
        alignRot.Mode = Enum.OrientationAlignmentMode.OneAttachment
        alignRot.Attachment0 = farmAttachment
        alignRot.RigidityEnabled = true
        alignRot.CFrame = rootPart.CFrame * CFrame.Angles(math.rad(90), 0, 0)
        alignRot.Parent = rootPart
    end

    alignPos.Position = targetPos
end

-- =======================================================
--   SERVER HOPPING ENGINE
-- =======================================================
local function serverHop()
    sendNotification({ Title = "Server Hopping", Desc = "Rival farmers detected! Finding fresh server..." })
    task.wait(1)
    queueScriptReload()

    task.spawn(function()
        local placeId = game.PlaceId
        local servers = {}
        local req = (syn and syn.request)
            or (http and http.request)
            or http_request
            or (fluxus and fluxus.request)
            or request

        if req then
            local ok, res = pcall(req, {
                Url    = "https://games.roblox.com/v1/games/" .. placeId .. "/servers/0?sortOrder=Asc&limit=100",
                Method = "GET"
            })
            if ok and res and res.Body then
                local ok2, body = pcall(function()
                    return game:GetService("HttpService"):JSONDecode(res.Body)
                end)
                if ok2 and body and body.data then
                    for _, v in ipairs(body.data) do
                        if type(v) == "table" and v.maxPlayers and v.playing and v.id then
                            if v.playing < v.maxPlayers and v.id ~= game.JobId then
                                table.insert(servers, v.id)
                            end
                        end
                    end
                end
            end
        end

        if #servers > 0 then
            TeleportService:TeleportToPlaceInstance(placeId, servers[math.random(1, #servers)], player)
        else
            TeleportService:Teleport(placeId, player)
        end
    end)
end

local function optimizePart(part)
    if part:IsA("BasePart") then
        part.Material = Enum.Material.SmoothPlastic
        part.CastShadow = false
    elseif part:IsA("ParticleEmitter") or part:IsA("Trail") or part:IsA("Sparkles") or part:IsA("Fire") or part:IsA("Smoke") then
        part.Enabled = false
    elseif part:IsA("Decal") or part:IsA("Texture") then
        part.Transparency = 1
    end
end

local function applyAntiLag()
    pcall(function()
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 9e9

        for _, effect in ipairs(Lighting:GetChildren()) do
            if effect:IsA("PostEffect") or effect:IsA("Atmosphere") or effect:IsA("BloomEffect")
            or effect:IsA("BlurEffect") or effect:IsA("SunRaysEffect")
            or effect:IsA("ColorCorrectionEffect") or effect:IsA("DepthOfFieldEffect") then
                effect.Enabled = false
            end
        end

        for _, desc in ipairs(Workspace:GetDescendants()) do
            optimizePart(desc)
        end

        if not antiLagConnection then
            antiLagConnection = Workspace.DescendantAdded:Connect(function(desc)
                if antiLagEnabled then
                    task.wait()
                    optimizePart(desc)
                end
            end)
        end
    end)
end

local function scanCoinContainers(callback)
    local checked = {}
    local containerNames = { "CoinContainer", "CoinVisuals", "Coins" }
    for _, name in ipairs(containerNames) do
        local container = Workspace:FindFirstChild(name, true)
        if container then
            for _, obj in ipairs(container:GetChildren()) do
                if not checked[obj] then
                    checked[obj] = true
                    callback(obj)
                end
            end
        end
    end
end

local function getSortedCoinPath()
    local coinPool = {}

    scanCoinContainers(function(obj)
        if obj:IsA("BasePart") and not coinBlacklists[obj] and obj.Parent then
            if obj.Name == "Coin_Server"
            or obj:FindFirstChildWhichIsA("TouchTransmitter")
            or obj:FindFirstChild("TouchInterest") then
                local visual = obj:FindFirstChild("CoinVisual")
                if (not visual or visual.Transparency == 0) and not isCoinTargetedByFarmer(obj) then
                    table.insert(coinPool, obj)
                end
            end
        end
    end)

    if #coinPool == 0 then
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("BasePart") and obj.Name == "Coin_Server"
            and not coinBlacklists[obj] and obj.Parent then
                if (obj:FindFirstChildWhichIsA("TouchTransmitter") or obj:FindFirstChild("TouchInterest"))
                and not isCoinTargetedByFarmer(obj) then
                    table.insert(coinPool, obj)
                end
            end
        end
    end

    if #coinPool == 0 then return {} end

    local sortedPath = {}
    local currentVirtualPos = rootPart and rootPart.Position or Vector3.zero

    while #coinPool > 0 do
        local nearestIdx, shortestDist = nil, math.huge
        for i, coin in ipairs(coinPool) do
            if coin and coin.Parent and not coinBlacklists[coin] then
                local dist = (coin.Position - currentVirtualPos).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    nearestIdx   = i
                end
            end
        end
        if nearestIdx then
            local chosen = coinPool[nearestIdx]
            table.insert(sortedPath, chosen)
            currentVirtualPos = chosen.Position
            table.remove(coinPool, nearestIdx)
        else
            break
        end
    end

    return sortedPath
end

local function getNearestValidCoin(hrp)
    local nearest, minDist = nil, math.huge
    scanCoinContainers(function(coin)
        if coin:IsA("BasePart") and not coinBlacklists[coin] and coin.Parent then
            if coin.Name == "Coin_Server" or coin:FindFirstChildWhichIsA("TouchTransmitter") then
                local visual = coin:FindFirstChild("CoinVisual")
                if (not visual or visual.Transparency == 0) and not isCoinTargetedByFarmer(coin) then
                    local dist = (hrp.Position - coin.Position).Magnitude
                    if dist < minDist then
                        minDist  = dist
                        nearest  = coin
                    end
                end
            end
        end
    end)
    return nearest
end

local function getNearestCoinToPos(pos)
    local nearest, minDist = nil, math.huge
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name == "Coin_Server" and obj.Parent then
            if obj:FindFirstChildWhichIsA("TouchTransmitter") or obj:FindFirstChild("TouchInterest") then
                local dist = (obj.Position - pos).Magnitude
                if dist < minDist then
                    minDist = dist
                    nearest = obj
                end
            end
        end
    end
    return nearest, minDist
end

-- =======================================================
--   RIVAL DETECTION
-- =======================================================
RunService.Heartbeat:Connect(function(dt)
    if not detectRivalFarmers or dt <= 0 then return end

    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude

    for _, targetPlr in ipairs(Players:GetPlayers()) do
        if targetPlr == player then continue end

        local tChar = targetPlr.Character
        local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
        local tHum  = tChar and tChar:FindFirstChildOfClass("Humanoid")

        if tRoot and tHum and tHum.Health > 0 then
            local currentPos = tRoot.Position
            local lastPos    = playerLastPositions[targetPlr]

            if lastPos then
                local moveVector = currentPos - lastPos
                local distMoved  = moveVector.Magnitude

                local horizontalSpeed = Vector3.new(moveVector.X, 0, moveVector.Z).Magnitude / dt
                local isSpeeding      = horizontalSpeed > 19.5

                local isNoclipping = false
                if distMoved > 1 and distMoved < 100 then
                    raycastParams.FilterDescendantsInstances = { tChar, character, Workspace:FindFirstChild("StormHub_AntiTiltFloor") }
                    local rayResult = Workspace:Raycast(lastPos, moveVector, raycastParams)
                    if rayResult and rayResult.Instance and rayResult.Instance.CanCollide then
                        isNoclipping = true
                    end
                end

                local isMovingToCoinStraight = false
                if distMoved > 1.5 then
                    local nearestCoin, coinDist = getNearestCoinToPos(lastPos)
                    if nearestCoin and coinDist < 150 then
                        local dotProduct = moveVector.Unit:Dot((nearestCoin.Position - lastPos).Unit)
                        if dotProduct > 0.96 then
                            isMovingToCoinStraight = true
                        end
                    end
                end

                if isSpeeding or isNoclipping or isMovingToCoinStraight then
                    playerSpeedFrames[targetPlr] = (playerSpeedFrames[targetPlr] or 0) + 1
                else
                    playerSpeedFrames[targetPlr] = math.max(0, (playerSpeedFrames[targetPlr] or 0) - 1)
                end

                if (playerSpeedFrames[targetPlr] or 0) >= 8 and not detectedAutoFarmers[targetPlr] then
                    detectedAutoFarmers[targetPlr] = true

                    local reasons = {}
                    if isSpeeding          then table.insert(reasons, string.format("Speed (%.1f)", horizontalSpeed)) end
                    if isNoclipping        then table.insert(reasons, "Noclip/Wallpass") end
                    if isMovingToCoinStraight then table.insert(reasons, "Linear Trajectory") end

                    sendNotification({
                        Title    = "⚠️ Rival Farmer Flagged!",
                        Desc     = targetPlr.Name .. ": " .. table.concat(reasons, ", "),
                        Duration = 4
                    })

                    pcall(function()
                        local hl = tChar:FindFirstChild("StormHub_FarmerHighlight") or Instance.new("Highlight")
                        hl.Name             = "StormHub_FarmerHighlight"
                        hl.FillColor        = Color3.fromRGB(255, 0, 50)
                        hl.OutlineColor     = Color3.fromRGB(255, 255, 255)
                        hl.FillTransparency = 0.3
                        hl.Parent           = tChar
                    end)

                    local count = 0
                    for _ in pairs(detectedAutoFarmers) do count += 1 end
                    if autoServerHopOnFarmers and count >= maxFarmersBeforeHop then
                        serverHop()
                    end
                end
            end

            playerLastPositions[targetPlr] = currentPos
        else
            playerLastPositions[targetPlr] = nil
            playerSpeedFrames[targetPlr]   = nil
        end
    end
end)

-- SILENT KNIFE THROW
function ExecuteSilentThrow()
    if not character or not rootPart then return end
    local knife = (character:FindFirstChild("Knife")) or (player.Backpack and player.Backpack:FindFirstChild("Knife"))
    if not knife then return end

    local events = knife:FindFirstChild("Events")
    local knifeThrown = events and events:FindFirstChild("KnifeThrown")

    local target, minDist = nil, 300
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local dist = (p.Character.HumanoidRootPart.Position - rootPart.Position).Magnitude
                if dist < minDist then
                    minDist = dist
                    target = p
                end
            end
        end
    end

    if target and target.Character then
        local tRoot = target.Character.HumanoidRootPart
        local tHum  = target.Character:FindFirstChildOfClass("Humanoid")
        local ping  = 0
        pcall(function() ping = Stats.Network.ServerStatsItem["Data Ping"]:GetValue() / 1000 end)

        local predTime = (State.Prediction * State.GunLeadMult) + ping
        local moveVec  = tRoot.AssemblyLinearVelocity
        if moveVec.Magnitude < 0.5 and tHum and tHum.MoveDirection.Magnitude > 0 then
            moveVec = tHum.MoveDirection * tHum.WalkSpeed
        end

        local predictedPos = tRoot.Position + (moveVec * predTime)

        if knife.Parent == player.Backpack and character:FindFirstChildOfClass("Humanoid") then
            character:FindFirstChildOfClass("Humanoid"):EquipTool(knife)
            task.wait(0.05)
        end

        if knifeThrown then
            knifeThrown:FireServer(CFrame.new(rootPart.Position), CFrame.new(predictedPos))
            sendNotification({ Title = "Silent Throw", Desc = "Knife thrown at " .. target.Name })
        end
    end
end

-- SILENT KNIFE AURA
function ToggleSilentKnifeAura(state)
    silentAuraEnabled = state
    if silentAuraConnection then silentAuraConnection:Disconnect() silentAuraConnection = nil end

    if state then
        silentAuraConnection = RunService.Heartbeat:Connect(function()
            if not silentAuraEnabled or getLocalRole() ~= "Murderer" then return end
            local knife = (character and character:FindFirstChild("Knife")) or (player.Backpack and player.Backpack:FindFirstChild("Knife"))
            if not knife then return end

            local events = knife:FindFirstChild("Events")
            local handleTouched = events and events:FindFirstChild("HandleTouched")
            local knifeStabbed = events and events:FindFirstChild("KnifeStabbed")

            if not handleTouched then return end

            local hitAny = false
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= player and p.Character then
                    local tRoot = p.Character:FindFirstChild("HumanoidRootPart")
                    local tHum  = p.Character:FindFirstChildOfClass("Humanoid")
                    if tRoot and tHum and tHum.Health > 0 then
                        if (tRoot.Position - rootPart.Position).Magnitude <= silentAuraRange then
                            pcall(function() handleTouched:FireServer(tRoot) end)
                            hitAny = true
                        end
                    end
                end
            end

            if hitAny and knifeStabbed then
                pcall(function() knifeStabbed:FireServer() end)
            end
        end)
    end
end

local function FindGunHolder()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == player then continue end
        local char = plr.Character
        if char and char:FindFirstChild("Gun") then return plr end
        if plr:FindFirstChild("Backpack") and plr.Backpack:FindFirstChild("Gun") then return plr end
    end
    return nil
end

function AutoKillTargets()
    if getLocalRole() ~= "Murderer" or not isActive then return end

    if priorityKillFarmers then
        for farmerPlr in pairs(detectedAutoFarmers) do
            if farmerPlr and farmerPlr.Parent and farmerPlr.Character then
                local fChar  = farmerPlr.Character
                local fRoot  = fChar:FindFirstChild("HumanoidRootPart")
                local fHum   = fChar:FindFirstChildOfClass("Humanoid")
                local myRoot = character:FindFirstChild("HumanoidRootPart")
                if fRoot and myRoot and fHum and fHum.Health > 0 then
                    fRoot.CanCollide = false
                    fRoot.CFrame     = myRoot.CFrame + myRoot.CFrame.LookVector * 2
                    local knife = EquipKnife()
                    if knife and knife:FindFirstChild("Stab") then
                        knife.Stab:FireServer("Down")
                    end
                    task.wait(0.15)
                end
            end
        end
    end

    local gunHolder = FindGunHolder()
    if gunHolder and gunHolder.Character then
        local tRoot  = gunHolder.Character:FindFirstChild("HumanoidRootPart")
        local tHum   = gunHolder.Character:FindFirstChildOfClass("Humanoid")
        local myRoot = character:FindFirstChild("HumanoidRootPart")
        if tRoot and myRoot and tHum and tHum.Health > 0 then
            tRoot.CanCollide = false
            tRoot.CFrame     = myRoot.CFrame + myRoot.CFrame.LookVector * 2
            local knife = EquipKnife()
            if knife and knife:FindFirstChild("Stab") then
                knife.Stab:FireServer("Down")
            end
        end
    end
end

function ShootMurderer()
    if not murdererPlayer or not murdererPlayer.Character then return end
    local mRoot  = murdererPlayer.Character:FindFirstChild("HumanoidRootPart")
    local myChar = player.Character
    if not mRoot or not myChar or not rootPart then return end

    local gun = myChar:FindFirstChild("Gun") or player.Backpack:FindFirstChild("Gun")
    if gun and gun.Parent == player.Backpack then
        local hum = myChar:FindFirstChildOfClass("Humanoid")
        if hum then
            hum:EquipTool(gun)
            task.wait(0.2)
        end
        gun = myChar:FindFirstChild("Gun")
    end
    if not gun then return end

    local shootRemote = gun:FindFirstChild("Shoot")
        or (GameplayRemotes and GameplayRemotes:FindFirstChild("ShootGun"))
    if not shootRemote then return end

    local originPos = mRoot.Position + Vector3.new(0, 0, -0.4)
    local ping = 0
    pcall(function()
        ping = Stats.Network.ServerStatsItem["Data Ping"]:GetValue() / 1000
    end)

    local timeOffset   = (State.Prediction * State.GunLeadMult) + ping
    local predictedPos = mRoot.Position + (mRoot.AssemblyLinearVelocity * timeOffset)
    local shotOrigin   = CFrame.lookAt(originPos, predictedPos)

    shootRemote:FireServer(shotOrigin, CFrame.new(predictedPos))
end

function GetGun()
    local attempts = 0
    isSearchingForGun = true

    while isActive and attempts < 5 do
        local myChar = player.Character
        local root   = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if not root then task.wait(0.1) continue end

        if player.Backpack:FindFirstChild("Gun") or (myChar and myChar:FindFirstChild("Gun")) then
            isSearchingForGun = false
            return true
        end

        local drop = Workspace:FindFirstChild("GunDrop", true)
        if drop and drop:IsA("BasePart") then
            root.AssemblyLinearVelocity = Vector3.zero
            drop.CFrame = root.CFrame + Vector3.new(0, 2, 0)
            task.wait(0.05)

            if firetouchinterest then
                firetouchinterest(root, drop, 0)
                firetouchinterest(root, drop, 1)
            end

            local prompt = drop:FindFirstChildOfClass("ProximityPrompt")
            if prompt then fireproximityprompt(prompt) end
            task.wait(0.8)
        else
            task.wait(1)
        end

        attempts += 1
    end

    isSearchingForGun = false
    return false
end

-- =======================================================
--   KINEMATIC PATH DANGER CHECK
-- =======================================================
local function isPathDangerous(startPos, endPos, murdPos)
    if ignoreMurderer or getLocalRole() == "Murderer" or murdererPlayer == player then return false end
    if not murdPos or not murdererPlayer or not murdererPlayer.Character then return false end

    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = { murdererPlayer.Character, character, safetyPlatform }

    if (endPos - murdPos).Magnitude <= 25 then
        local ray = Workspace:Raycast(murdPos, endPos - murdPos, rp)
        drawVisualTrace(murdPos, endPos, Color3.fromRGB(255, 50, 50),
            string.format("Info - %.1f studs\nWhy: Destination Proximity", (endPos - murdPos).Magnitude), 0.8)
        if not ray then return true end
    end

    local pathVec    = endPos - startPos
    local pathLen    = pathVec.Magnitude
    if pathLen == 0 then return false end

    local pathDir    = pathVec.Unit
    local toMurd     = murdPos - startPos
    local projection = toMurd:Dot(pathDir)

    if projection > 0 and projection < pathLen then
        local closestPoint = startPos + pathDir * projection
        if (murdPos - closestPoint).Magnitude <= 25 then
            local ray2 = Workspace:Raycast(murdPos, closestPoint - murdPos, rp)
            drawVisualTrace(murdPos, closestPoint, Color3.fromRGB(255, 130, 0),
                string.format("Info - %.1f studs\nWhy: Vector Path Intersect", (closestPoint - murdPos).Magnitude), 0.8)
            if not ray2 then return true end
        end
    end

    return false
end

-- =======================================================
--   MOVEMENT HELPER
-- =======================================================
local function moveToPosition(targetPos, speed)
    if not rootPart or not isActive then return end
    rootPart.Anchored = false

    local distance = (targetPos - rootPart.Position).Magnitude
    if distance < 0.3 then return end

    if distance >= 200 then
        drawVisualTrace(rootPart.Position, targetPos, Color3.fromRGB(0, 130, 255),
            string.format("TP - %.1f studs", distance), 1.5)
        rootPart.CFrame               = CFrame.new(targetPos)
        rootPart.AssemblyLinearVelocity  = Vector3.zero
        rootPart.AssemblyAngularVelocity = Vector3.zero
        task.wait(0.05)
        return
    end

    local duration = distance / (speed or flySpeed)
    local tween    = TweenService:Create(rootPart,
        TweenInfo.new(duration, Enum.EasingStyle.Linear),
        { CFrame = CFrame.new(targetPos) })
    tween:Play()

    local cancelCheck
    cancelCheck = RunService.Heartbeat:Connect(function()
        if not isActive then
            tween:Cancel()
            if cancelCheck then cancelCheck:Disconnect() end
        end
        if rootPart then
            rootPart.AssemblyLinearVelocity  = Vector3.zero
            rootPart.AssemblyAngularVelocity = Vector3.zero
        end
    end)

    tween.Completed:Wait()
    if cancelCheck and cancelCheck.Connected then cancelCheck:Disconnect() end
    if rootPart then
        rootPart.AssemblyLinearVelocity  = Vector3.zero
        rootPart.AssemblyAngularVelocity = Vector3.zero
    end
end

local function stopFarm()
    isActive         = false
    isRoundActive    = false
    isSearchingForGun = false
    cleanupAlignFarm()
    Workspace.Gravity = originalGravity
    if rootPart then
        rootPart.Anchored                = false
        rootPart.AssemblyLinearVelocity  = Vector3.zero
        rootPart.CFrame                  = CFrame.new(safeWaitingPlace)
    end
end

local function startFarm()
    if isActive then return end
    isActive = true

    originalGravity                    = Workspace.Gravity
    Workspace.Gravity                  = 0
    Workspace.FallenPartsDestroyHeight = -50000

    if rootPart then rootPart.Anchored = false end
    if startTime == 0 then startTime = tick() end
    coinBlacklists = {}

    sendNotification({ Title = "Storm Hub Auto Farm", Desc = "AFK farm loop engaged." })

    task.spawn(function()
        while isActive do
            if getLocalRole() == "Murderer" then
                AutoKillTargets()
            end
            task.wait()
        end
    end)

    task.spawn(function()
        while isActive do
            if autoGrabGun then
                local myChar = player.Character
                local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
                local myHum  = myChar and myChar:FindFirstChildOfClass("Humanoid")

                if myRoot and myHum and myHum.Health > 0 then
                    local hasGun  = player.Backpack:FindFirstChild("Gun") or myChar:FindFirstChild("Gun")
                    local isMurd  = (getLocalRole() == "Murderer")

                    if not hasGun and not isMurd then
                        local drop = Workspace:FindFirstChild("GunDrop", true)
                        if drop and drop:IsA("BasePart") then
                            isSearchingForGun = true
                            drop.CFrame = myRoot.CFrame + Vector3.new(0, 1.5, 0)

                            if firetouchinterest then
                                firetouchinterest(myRoot, drop, 0)
                                firetouchinterest(myRoot, drop, 1)
                            end

                            local prompt = drop:FindFirstChildOfClass("ProximityPrompt")
                            if prompt then fireproximityprompt(prompt) end

                            task.wait(0.08)
                            local acquired = player.Backpack:FindFirstChild("Gun") or myChar:FindFirstChild("Gun")
                            if acquired then
                                sendNotification({ Title = "Gun Intercepted", Desc = "Pulled gun to character safely!" })
                            end
                            isSearchingForGun = false
                        end
                    end
                end
            end
            task.wait(0.1)
        end
    end)

    task.spawn(function()
        while isActive do
            character = player.Character or player.CharacterAdded:Wait()
            local humanoid = character:WaitForChild("Humanoid")
            rootPart = character:FindFirstChild("HumanoidRootPart")
            if not rootPart or humanoid.Health <= 0 then task.wait(0.1) continue end

            rootPart.Anchored = false

            if isPlayerInLobby() then
                localCoins     = 0
                coinBlacklists = {}
                cleanupAlignFarm()
                rootPart.AssemblyLinearVelocity = Vector3.zero
                task.wait(0.5)
                continue
            end

            if isSearchingForGun then
                cleanupAlignFarm()
                if (rootPart.Position - safeWaitingPlace).Magnitude > 5 then
                    rootPart.CFrame = CFrame.new(safeWaitingPlace)
                end
                task.wait(0.2)
                continue
            end

            if bagsFull or localCoins >= maxCoinLimit then
                cleanupAlignFarm()
                local role = getLocalRole()

                if role == "Murderer" then
                    if not hasResetForFullBag then
                        hasResetForFullBag = true
                        moveToPosition(rootPart.Position + Vector3.new(0, 8, 0), 100)
                        task.wait(0.1)
                        KillAll()
                    else
                        task.wait(1)
                    end

                elseif role == "Sheriff" or role == "Hero" then
                    if not hasResetForFullBag then
                        if murdererPlayer and murdererPlayer.Character
                        and murdererPlayer.Character:FindFirstChild("HumanoidRootPart") then
                            ShootMurderer()
                            local mHum = murdererPlayer.Character:FindFirstChildOfClass("Humanoid")
                            if mHum and mHum.Health > 0 and getLocalRole() == role then
                                task.wait(1)
                            else
                                hasResetForFullBag = true
                                sendNotification({ Title = "Target Down", Desc = "Murderer eliminated." })
                            end
                        else
                            task.wait(0.5)
                        end
                    else
                        task.wait(1)
                    end

                else 
                    if not hasResetForFullBag then
                        local drop = Workspace:FindFirstChild("GunDrop", true)
                        if drop then
                            GetGun()
                        else
                            hasResetForFullBag = true
                            rootPart.CFrame    = CFrame.new(safeWaitingPlace)
                            task.wait(2)
                        end
                    else
                        task.wait(1)
                    end
                end
                continue
            end

            -- -----------------------------------------------
            --  FEATURE 1: ALIGNPOSITION FARM MODE
            -- -----------------------------------------------
            if farmMode == "AlignPosition" then
                local coinPath = getSortedCoinPath()
                if #coinPath == 0 then
                    cleanupAlignFarm()
                    if (rootPart.Position - safeWaitingPlace).Magnitude > 5 then
                        rootPart.CFrame = CFrame.new(safeWaitingPlace)
                    end
                    task.wait(0.2)
                    continue
                end

                for _, targetCoin in ipairs(coinPath) do
                    if not isActive or bagsFull or isSearchingForGun
                    or not targetCoin or not targetCoin.Parent
                    or coinBlacklists[targetCoin] then
                        break
                    end

                    local targetPos        = targetCoin.Position + Vector3.new(0, -4, 0)
                    local distanceToTarget = (targetPos - rootPart.Position).Magnitude

                    -- Murderer avoidance (Skipped if ignoreMurderer is TRUE)
                    if not ignoreMurderer and getLocalRole() ~= "Murderer" and distanceToTarget < 200
                    and murdererPosition and isPathDangerous(rootPart.Position, targetPos, murdererPosition) then
                        cleanupAlignFarm()
                        if (rootPart.Position - safeWaitingPlace).Magnitude > 3 then
                            rootPart.CFrame = CFrame.new(safeWaitingPlace)
                        end
                        task.wait(0.5)
                        break
                    end

                    moveToCoinAlign(targetPos)

                    local collectTimeout = tick()
                    while isActive and targetCoin and targetCoin.Parent and not coinBlacklists[targetCoin] do
                        if (targetCoin.Position - rootPart.Position).Magnitude <= 5 then
                            if firetouchinterest then
                                pcall(function()
                                    firetouchinterest(rootPart, targetCoin, 0)
                                    firetouchinterest(rootPart, targetCoin, 1)
                                end)
                            end
                            coinBlacklists[targetCoin] = true
                            break
                        end
                        if tick() - collectTimeout > 3 then break end
                        task.wait()
                    end
                end
                task.wait()
                continue
            else
                cleanupAlignFarm()
            end

            -- -----------------------------------------------
            --  VELOCITY (GLIDE) MODE
            -- -----------------------------------------------
            if farmMode == "Velocity (Glide)" then
                local targetCoin = getNearestValidCoin(rootPart)
                if targetCoin then
                    for _, part in ipairs(character:GetDescendants()) do
                        if part:IsA("BasePart") then part.CanCollide = false end
                    end

                    local targetPos = targetCoin.Position
                    local direction = (targetPos - rootPart.Position).Unit
                    rootPart.AssemblyLinearVelocity = direction * flySpeed
                    rootPart.CFrame = CFrame.new(rootPart.Position, targetPos)

                    if (targetPos - rootPart.Position).Magnitude < 3.5 then
                        if firetouchinterest then
                            local touch = targetCoin:FindFirstChildWhichIsA("TouchTransmitter") or targetCoin
                            firetouchinterest(rootPart, touch.Parent or touch, 0)
                            task.spawn(function()
                                task.wait()
                                firetouchinterest(rootPart, touch.Parent or touch, 1)
                            end)
                        end
                        coinBlacklists[targetCoin] = true
                    end
                else
                    rootPart.AssemblyLinearVelocity = Vector3.zero
                    task.wait(0.2)
                end
                task.wait()
                continue
            end

            -- -----------------------------------------------
            --  LERP + INSTANT TP MODES
            -- -----------------------------------------------
            local coinPath = getSortedCoinPath()

            if #coinPath == 0 then
                if (rootPart.Position - safeWaitingPlace).Magnitude > 5 then
                    rootPart.CFrame = CFrame.new(safeWaitingPlace)
                end
                task.wait(0.2)
                continue
            end

            for _, targetCoin in ipairs(coinPath) do
                if not isActive or bagsFull or isSearchingForGun
                or not targetCoin or not targetCoin.Parent
                or coinBlacklists[targetCoin] then
                    break
                end

                local targetPos         = targetCoin.Position + Vector3.new(0, -4, 0)
                local distanceToTarget  = (targetPos - rootPart.Position).Magnitude

                -- Murderer avoidance (Skipped if ignoreMurderer is TRUE)
                if not ignoreMurderer and getLocalRole() ~= "Murderer" and distanceToTarget < 200
                and murdererPosition and isPathDangerous(rootPart.Position, targetPos, murdererPosition) then
                    if (rootPart.Position - safeWaitingPlace).Magnitude > 3 then
                        rootPart.CFrame = CFrame.new(safeWaitingPlace)
                    end
                    task.wait(0.5)
                    break
                end

                if farmMode == "Instant TP" then
                    drawVisualTrace(rootPart.Position, targetPos, Color3.fromRGB(0, 130, 255),
                        string.format("TP Coin - %.1f studs", distanceToTarget), 0.4)
                    rootPart.CFrame               = CFrame.new(targetPos)
                    rootPart.AssemblyLinearVelocity  = Vector3.zero
                    rootPart.AssemblyAngularVelocity = Vector3.zero
                else 
                    if distanceToTarget >= 200 then
                        drawVisualTrace(rootPart.Position, targetPos, Color3.fromRGB(0, 130, 255),
                            string.format("TP - %.1f studs", distanceToTarget), 1.5)
                        rootPart.CFrame               = CFrame.new(targetPos)
                        rootPart.AssemblyLinearVelocity  = Vector3.zero
                        rootPart.AssemblyAngularVelocity = Vector3.zero
                        task.wait(0.05)
                    else
                        drawVisualTrace(rootPart.Position, targetPos, Color3.fromRGB(0, 255, 127),
                            string.format("Coin - %.1f studs", distanceToTarget), distanceToTarget / flySpeed)
                        moveToPosition(targetPos, flySpeed)
                    end
                end

                if targetCoin and targetCoin.Parent and not coinBlacklists[targetCoin] and isActive then
                    if firetouchinterest then
                        pcall(function()
                            firetouchinterest(rootPart, targetCoin, 0)
                            firetouchinterest(rootPart, targetCoin, 1)
                        end)
                    end
                    RunService.Heartbeat:Wait()
                    coinBlacklists[targetCoin] = true

                    if farmMode == "Instant TP" then task.wait(0.5) end
                end
            end
        end
    end)
end

local function updateMatchData()
    local success, data = pcall(function() return GetPlayerDataRemote:InvokeServer() end)
    if success and type(data) == "table" then
        matchData = data
        local myData = matchData[player.Name]
        if myData then
            localCoins = myData.Coins or 0
            bagsFull   = (localCoins >= maxCoinLimit)
            if localCoins < maxCoinLimit then hasResetForFullBag = false end
        end

        murdererPosition = nil
        murdererPlayer   = nil
        for name, info in pairs(matchData) do
            if info.Role == "Murderer" then
                local plr = Players:FindFirstChild(name)
                if plr and plr.Character then
                    murdererPlayer = plr
                    local root = plr.Character:FindFirstChild("HumanoidRootPart")
                    if root then murdererPosition = root.Position end
                end
                break
            end
        end
    end

    if autoOpenCrate and #selectedCrates > 0 and playerGui:FindFirstChild("CrossPlatform") then
        pcall(function()
            local amountLabel = playerGui.CrossPlatform.Shop.Small.Container.Title.Container.Coins.Container.Amount
            local walletCoins = tonumber(string.gsub(amountLabel.Text, ",", ""))

            if walletCoins then
                if currentCrateIndex > #selectedCrates then currentCrateIndex = 1 end
                local targetCrate = selectedCrates[currentCrateIndex]
                if targetCrate and walletCoins >= 1000 then
                    OpenCrateRemote:InvokeServer(targetCrate, "MysteryBox", targetType)
                    currentCrateIndex += 1
                    if currentCrateIndex > #selectedCrates then currentCrateIndex = 1 end
                    task.wait(0.3)
                end
            end
        end)
    end
end

task.spawn(function()
    while task.wait(1) do
        if isActive then updateMatchData() end
    end
end)
